[mcp]
# Format: mcp:<server>  — uses wildcard for all tools from that server
# Format: mcp:<server>:read  — only read/get/list/search tools (no writes)
#
# Claude server name convention: mcp__<server>__<tool>
# opencode: auto-allows all configured MCP servers (no explicit entry needed)
# Antigravity: uses mcp(<server>/*) — only applies if server is configured there
#
# Wildcard servers (all tools trusted):
mcp:buildkite-ro
mcp:elastic-docs
mcp:elastic-team-internal-docs
#
# Read-only servers (specific safe tools listed in sync script):
mcp:slack:read
mcp:atlassian:read
mcp:google-calendar:read
